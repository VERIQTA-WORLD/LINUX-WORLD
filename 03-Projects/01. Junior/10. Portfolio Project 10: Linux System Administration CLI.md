# PHASE 2: Build the CLI Foundation

Before building a large script, understand the CLI pattern.

A user runs:

```bash

```

Bash receives:

```text
$0 = sysadminctl
$1 = storage
```

If the user runs:

```bash
sysadminctl processes --limit 10
```

then conceptually:

```text
$1 = processes
$2 = --limit
$3 = 10
```

We need to translate arguments into actions.

```bash
#!/usr/bin/env bash

SYSADMINCTL_VERSION="1.0.0"

print_line() {
    printf '%*s\n' 60 '' | tr ' ' '='
}

print_subline() {
    printf '%*s\n' 60 '' | tr ' ' '-'
}

section() {
    echo
    print_line
    printf ' %s\n' "$1"
    print_line
    echo
}

subsection() {
    echo
    print_subline
    printf ' %s\n' "$1"
    print_subline
}

info() {
    printf '[INFO] %s\n' "$1"
}

ok() {
    printf '[OK]   %s\n' "$1"
}

warn() {
    printf '[WARN] %s\n' "$1"
}

critical() {
    printf '[CRIT] %s\n' "$1"
}

error() {
    printf '[ERROR] %s\n' "$1" >&2
}

command_exists() {
    command -v "$1" >/dev/null 2>&1
}

require_command() {

    local command_name="$1"

    if ! command_exists "$command_name"; then
        error "Required command not found: $command_name"
        return 1
    fi

    return 0
}

is_integer() {
    [[ "$1" =~ ^[0-9]+$ ]]
}
```

```bash
show_system_info() {
```

# PHASE 5: Build Uptime Function

Add to `system.sh`:

```bash
show_uptime() {

    section "UPTIME"

    uptime

    echo

    if [[ -r /proc/uptime ]]; then

        local uptime_seconds
        local days
        local hours
        local minutes

        uptime_seconds="$(
            awk '{print int($1)}' /proc/uptime
        )"

        days=$((uptime_seconds / 86400))
        hours=$(((uptime_seconds % 86400) / 3600))
        minutes=$(((uptime_seconds % 3600) / 60))

        printf 'Calculated uptime: %d days, %d hours, %d minutes\n' \
            "$days" \
            "$hours" \
            "$minutes"
    fi
}
```

### Why `/proc/uptime`?

The Linux kernel exposes runtime information through the `/proc` pseudo-filesystem.

Run manually:

```bash
cat /proc/uptime
```

Representative:

```text
38412.45 142321.30
```

The first value represents system uptime in seconds.

# PHASE 6: Build CPU Information

Add:

```bash
show_cpu() {

    section "CPU INFORMATION"

    local cpu_model
    local logical_cpus
    local load1
    local load5
    local load15

    cpu_model="$(
        awk -F: \
        '/model name/ {
            gsub(/^[ \t]+/, "", $2)
            print $2
            exit
        }' \
        /proc/cpuinfo
    )"

    logical_cpus="$(
        getconf _NPROCESSORS_ONLN
    )"

    read -r load1 load5 load15 _ < /proc/loadavg

    printf '%-22s %s\n' \
        "CPU model:" \
        "${cpu_model:-unknown}"

    printf '%-22s %s\n' \
        "Logical CPUs:" \
        "$logical_cpus"

    printf '%-22s %s\n' \
        "1-minute load:" \
        "$load1"

    printf '%-22s %s\n' \
        "5-minute load:" \
        "$load5"

    printf '%-22s %s\n' \
        "15-minute load:" \
        "$load15"

    subsection "Top CPU-Consuming Processes"

    ps \
        -eo pid,user,%cpu,%mem,comm \
        --sort=-%cpu \
    | head -n 11
}
```


Load average is **not simply CPU percentage**.

On Linux, it reflects runnable tasks and tasks in uninterruptible sleep.

# PHASE 7: Build Memory Information

Add:

```bash
show_memory() {

    section "MEMORY INFORMATION"

    free -h

    echo

    local total_kb
    local available_kb
    local used_kb
    local used_percent

    total_kb="$(
        awk \
        '/MemTotal:/ {
            print $2
        }' \
        /proc/meminfo
    )"

    available_kb="$(
        awk \
        '/MemAvailable:/ {
            print $2
        }' \
        /proc/meminfo
    )"

    if [[ -n "$total_kb" ]] &&
       [[ -n "$available_kb" ]] &&
       (( total_kb > 0 )); then

        used_kb=$((total_kb - available_kb))

        used_percent=$(
            awk \
            -v used="$used_kb" \
            -v total="$total_kb" \
            'BEGIN {
                printf "%.1f", (used / total) * 100
            }'
        )

        printf '\nEstimated memory used: %s%%\n' \
            "$used_percent"
    fi

    subsection "Top Memory-Consuming Processes"

    ps \
        -eo pid,user,%mem,%cpu,comm \
        --sort=-%mem \
    | head -n 11
}
```

---

## Why Use `MemAvailable`?

A common beginner mistake is treating all non-free RAM as unavailable.

Linux intentionally uses memory for:

```text
Page cache
Buffers
Kernel structures
Applications
```

`MemAvailable` provides a more useful estimate of memory that can be made available for new workloads without swapping.

```bash
show_processes() {

    local limit="${1:-10}"

    if ! is_integer "$limit"; then
        error "Process limit must be an integer."
        return 2
    fi

    if (( limit < 1 || limit > 100 )); then
        error "Process limit must be between 1 and 100."
        return 2
    fi

    section "PROCESS INFORMATION"

    printf '%-24s %s\n' \
        "Total processes:" \
        "$(ps -e --no-headers | wc -l)"

    printf '%-24s %s\n' \
        "Running processes:" \
        "$(ps -eo stat= | grep -c '^R' || true)"

    printf '%-24s %s\n' \
        "Zombie processes:" \
        "$(ps -eo stat= | grep -c '^Z' || true)"

    subsection "Top CPU Processes"

    ps \
        -eo pid,ppid,user,stat,%cpu,%mem,etime,comm \
        --sort=-%cpu \
    | head -n $((limit + 1))

    subsection "Top Memory Processes"

    ps \
        -eo pid,ppid,user,stat,%mem,%cpu,etime,comm \
        --sort=-%mem \
    | head -n $((limit + 1))
}
```

```bash
show_network() {

    section "NETWORK INFORMATION"

    subsection "Interfaces"

    ip -brief address

    subsection "Routes"

    ip route

    subsection "Default Route"

    local default_route

    default_route="$(
        ip route show default |
        head -n 1
    )"

    if [[ -n "$default_route" ]]; then
        printf '%s\n' "$default_route"
    else
        warn "No default route detected."
    fi

    subsection "Listening TCP/UDP Sockets"

    if (( EUID == 0 )); then
        ss -lntup
    else
        ss -lntu
        echo
        info "Run as root to request additional process details where available."
    fi

    subsection "DNS Resolver Configuration"

    if [[ -r /etc/resolv.conf ]]; then

        grep \
            -E '^(nameserver|search|domain)' \
            /etc/resolv.conf \
        || true

    else
        warn "/etc/resolv.conf is not readable."
    fi
}
```

# PHASE 12: Build Log Information

Add:

```bash
show_logs() {

    local lines="${1:-20}"

    if ! is_integer "$lines"; then
        error "Log line count must be an integer."
        return 2
    fi

    if (( lines < 1 || lines > 200 )); then
        error "Log line count must be between 1 and 200."
        return 2
    fi

    section "RECENT HIGH-PRIORITY LOGS"

    if ! command_exists journalctl; then
        error "journalctl is not available."
        return 1
    fi

    journalctl \
        -b \
        -p err \
        -n "$lines" \
        --no-pager
}
```

### What does this do?

```text
-b
    Current boot

-p err
    Error priority and more severe messages

-n N
    Last N matching entries

--no-pager
    Print directly instead of opening pager
```

### Permission Note

A normal user may not have access to every journal entry.

The CLI should not automatically run itself through `sudo`.

Instead, document when elevated access may reveal additional information.

# PHASE 17: Memory Health Check

Add:

```bash
check_memory_health() {

    local total_kb
    local available_kb
    local used_kb
    local used_percent

    total_kb="$(
        awk \
        '/MemTotal:/ {
            print $2
        }' \
        /proc/meminfo
    )"

    available_kb="$(
        awk \
        '/MemAvailable:/ {
            print $2
        }' \
        /proc/meminfo
    )"

    if [[ -z "$total_kb" ]] ||
       [[ -z "$available_kb" ]] ||
       (( total_kb <= 0 )); then

        health_warn "Unable to calculate memory usage."
        return
    fi

    used_kb=$((total_kb - available_kb))

    used_percent=$(
        awk \
        -v used="$used_kb" \
        -v total="$total_kb" \
        'BEGIN {
            printf "%d", (used / total) * 100
        }'
    )

    if (( used_percent >= MEMORY_CRIT_PERCENT )); then

        health_crit \
        "Memory usage ${used_percent}% >= critical threshold ${MEMORY_CRIT_PERCENT}%"

    elif (( used_percent >= MEMORY_WARN_PERCENT )); then

        health_warn \
        "Memory usage ${used_percent}% >= warning threshold ${MEMORY_WARN_PERCENT}%"

    else

        health_ok \
        "Memory usage ${used_percent}%"

    fi
}
```

# PHASE 18: CPU Load Health

CPU load needs context.

A load of:

```text
4.0
```

```text
2 CPUs
```

than on:

```text
16 CPUs
```

```text

```

Add:

```bash
check_cpu_health() {

    local load1
    local cpus
    local load_percent

    read -r load1 _ < /proc/loadavg

    cpus="$(
        getconf _NPROCESSORS_ONLN
    )"

    if ! is_integer "$cpus" ||
       (( cpus < 1 )); then

        health_warn "Unable to determine online CPU count."
        return
    fi

    load_percent=$(
        awk \
        -v load="$load1" \
        -v cpus="$cpus" \
        'BEGIN {
            printf "%d", (load / cpus) * 100
        }'
    )

    if (( load_percent >= CPU_LOAD_CRIT_PERCENT )); then

        health_crit \
        "1-minute load equivalent ${load_percent}% of online CPU count"

    elif (( load_percent >= CPU_LOAD_WARN_PERCENT )); then

        health_warn \
        "1-minute load equivalent ${load_percent}% of online CPU count"

    else

        health_ok \
        "1-minute load equivalent ${load_percent}% of online CPU count"

    fi
}
```

### Important

This is a **heuristic**, not CPU utilization.

The tool deliberately calls it:



# PHASE 24: Gateway Reachability

A subtle point:

The presence of a default route does not prove the gateway is reachable.

Add:

```bash
check_gateway() {

    if [[ "$CHECK_GATEWAY" != "true" ]]; then
        return
    fi

    local gateway

    gateway="$(
        ip route show default |
        awk '
            /^default/ {
                for (i = 1; i <= NF; i++) {
                    if ($i == "via") {
                        print $(i + 1)
                        exit
                    }
                }
            }
        '
    )"

    if [[ -z "$gateway" ]]; then

        health_warn \
        "Gateway address could not be determined"

        return
    fi

    if command_exists ping; then

        if ping \
            -c 1 \
            -W 2 \
            "$gateway" \
            >/dev/null 2>&1; then

            health_ok \
            "Default gateway $gateway responded to ICMP"

        else

            health_warn \
            "Default gateway $gateway did not respond to ICMP"

        fi

    else

        health_warn \
        "ping unavailable; gateway reachability not tested"

    fi
}
```

### Important

A failed ping does not always mean the gateway is down.

ICMP may be filtered.

The tool reports exactly what it knows:

> The gateway did not respond to ICMP.

It does not incorrectly claim:

> The gateway is offline.

That distinction is part of good diagnostics.

# 15. Security

# PHASE 33: Do Not Automatically Run Everything as Root

The CLI should work primarily as an ordinary user.

Why?

Because a diagnostic tool does not automatically need unlimited privilege.

Some commands provide additional information when run with elevated permissions.

For example:

```bash
sudo ss -lntup
```

may expose process information unavailable to a normal user.

Similarly, journal visibility can depend on permissions.

The correct model is:

```text
Normal mode
    ↓
Gather what user can access

Elevated mode when deliberately requested
    ↓
Gather privileged information
```

not:

```text
Everything runs as root by default
```

---

# PHASE 34: Protect Configuration

Check:

```bash
stat -c '%U %G %a %n' \
~/.config/sysadminctl/sysadminctl.conf
```

Expected owner:

```text
Current user
```

Recommended mode:

```text
600
```

# PHASE 35: Never Execute User Input

Suppose someone runs:

```bash
sysadminctl processes --limit '$(dangerous-command)'
```

The CLI should treat it as invalid text.

Our validation:

```bash
is_integer "$limit"
```

requires:

```text
digits only
```

and does not use:

```bash
eval
```

Never solve argument handling using `eval`.

---

# PHASE 36: Avoid Parsing Human Output When Better Interfaces Exist

Where practical, prefer structured or predictable command output.

Examples:

```bash
systemctl show
```

can be easier to process than parsing decorative:

```bash
systemctl status
```

Similarly:

```bash
ip -brief address
```

provides compact network information.

This is an important CLI-development principle.

Notice that the generic CLI test does **not** use:

# PHASE 41: Add Configuration Validation

Add to `common.sh`:

```bash
validate_percentage() {

    local name="$1"
    local value="$2"

    if ! is_integer "$value"; then
        error "$name must be an integer: $value"
        return 1
    fi

    if (( value < 0 || value > 100 )); then
        error "$name must be between 0 and 100: $value"
        return 1
    fi

    return 0
}
```

CPU load thresholds can exceed 100, so add:

```bash
validate_nonnegative_integer() {

    local name="$1"
    local value="$2"

    if ! is_integer "$value"; then
        error "$name must be a non-negative integer: $value"
        return 1
    fi

    return 0
}
```

Add:

```bash
validate_config() {

    local failed=0

    validate_nonnegative_integer \
        "CPU_LOAD_WARN_PERCENT" \
        "$CPU_LOAD_WARN_PERCENT" ||
        failed=1

    validate_nonnegative_integer \
        "CPU_LOAD_CRIT_PERCENT" \
        "$CPU_LOAD_CRIT_PERCENT" ||
        failed=1

    validate_percentage \
        "MEMORY_WARN_PERCENT" \
        "$MEMORY_WARN_PERCENT" ||
        failed=1

    validate_percentage \
        "MEMORY_CRIT_PERCENT" \
        "$MEMORY_CRIT_PERCENT" ||
        failed=1

    validate_percentage \
        "DISK_WARN_PERCENT" \
        "$DISK_WARN_PERCENT" ||
        failed=1

    validate_percentage \
        "DISK_CRIT_PERCENT" \
        "$DISK_CRIT_PERCENT" ||
        failed=1

    validate_percentage \
        "INODE_WARN_PERCENT" \
```

# FAILURE LAB 5: Warning Threshold Greater Than Critical

There is another logical problem.

This configuration is numerically valid:

```text
MEMORY_WARN_PERCENT=95
MEMORY_CRIT_PERCENT=80
```

But operationally it makes little sense.

Add to `validate_config()`:

```bash
if (( MEMORY_WARN_PERCENT >= MEMORY_CRIT_PERCENT )); then
    error "Memory warning threshold must be lower than critical threshold."
    failed=1
fi

if (( DISK_WARN_PERCENT >= DISK_CRIT_PERCENT )); then
    error "Disk warning threshold must be lower than critical threshold."
    failed=1
fi

if (( INODE_WARN_PERCENT >= INODE_CRIT_PERCENT )); then
    error "Inode warning threshold must be lower than critical threshold."
    failed=1
fi

if (( CPU_LOAD_WARN_PERCENT >= CPU_LOAD_CRIT_PERCENT )); then
    error "CPU load warning threshold must be lower than critical threshold."
    failed=1
fi
```

Now test deliberately.

This introduces a broader software-engineering lesson:

```text
Type validation
    ≠
Semantic validation
```

`95` is a valid integer.

That does not mean:

```text
warning=95
critical=80
```

is a sensible configuration.

# PHASE 44: Understand the Development Symlink Trade-Off

The system-wide command currently points into:


That is convenient for a development lab because edits immediately affect the installed command.

It is not ideal application packaging.

```text
Libraries:
/usr/local/lib/sysadminctl/

Configuration:
/etc/sysadminctl/
or
~/.config/sysadminctl/
```

Packaging and installers are good future improvements.

```bash
_sysadminctl_completion() {
```

# 23. Interview Preparation

## What is a CLI?

A command-line interface allows users to interact with software by entering commands and arguments in a terminal.

---

## What is Bash?

Bash is a Unix shell and scripting language commonly used for command execution and system automation.

---

## What does `$0` contain?

Usually the invoked script or command name.

---

## What does `$1` contain?

The first positional argument.

---

## What does `$@` represent?

All positional arguments while preserving argument boundaries when quoted as:

```bash
"$@"
```

---

## Why quote variables?

To prevent unintended word splitting and pathname expansion.

Prefer:

```bash
"$variable"
```

rather than:

```bash
$variable
```

unless unquoted expansion is deliberately required.

---

## Why use functions?

Functions provide:

```text
Reuse
Organization
Readability
Testing boundaries
Reduced duplication
```

---

## Why use `case` for command dispatch?

It provides a clear way to map command names to actions.

---

## What is an exit status?

A numeric value returned by a process to its caller.

Traditionally:

```text
0 = success
nonzero = some other condition
```

The meaning of nonzero values depends on the program's documented contract.

---

## Why did health return nonzero without the CLI being broken?

Because the health command uses exit status to communicate detected system state.

For this project:

```text
0 healthy
1 warning
2 critical
```

---

## What does `set -u` do?

It treats expansion of unset variables as an error in many contexts.

It can catch mistakes but requires careful scripting.

---

## What does `set -o pipefail` do?

A pipeline can return failure when an earlier pipeline component fails rather than blindly returning only the final command's status.

---

## Why didn't the project simply use `set -e` everywhere?

`set -e` has subtle behavior around conditionals, pipelines, functions, and expected failures.

For an educational diagnostic CLI, explicit error handling is often easier to reason about.

---

## What is `/proc`?

A virtual filesystem exposing kernel and process information.

Examples:

```text
/proc/cpuinfo
/proc/meminfo
/proc/loadavg
/proc/uptime
```

---

## What is `/proc/meminfo`?

A kernel-provided view of memory statistics.

---

## Why use `MemAvailable`?

It provides a useful estimate of memory available for new workloads without requiring swap, accounting for reclaimable memory better than simply looking at `MemFree`.

---

## What is load average?

A measure related to tasks that are runnable or in uninterruptible sleep over time.

---

## Is load average CPU utilization?

No.

---

## Why normalize load against CPU count?

A load of 4 has different implications on a two-CPU system and a sixteen-CPU system.

---

## What does `df` show?

Filesystem space usage.

---

## What does `df -i` show?

Filesystem inode usage.

---

## Why monitor inodes?

A filesystem can run out of inodes even when storage bytes remain available.

---

## What does `lsblk` show?

Block-device topology and metadata.

---

## What does `findmnt` show?

Mounted filesystem information.

---

## What does `ps` show?

Process information.

---

## How did the CLI identify top CPU processes?

Using:

```bash
ps \
-eo pid,user,%cpu,%mem,comm \
--sort=-%cpu
```

and limiting the output.

---

## What does `ip -brief address` show?

Compact interface and IP-address information.

---

## What does `ip route` show?

The kernel routing table.

---

## What is a default route?

The route used when no more specific route matches a destination.

---

## Does a default route prove Internet connectivity?

No.

It only proves the route is configured.

---

## Does a failed gateway ping prove the gateway is down?

No.

ICMP may be blocked or filtered.

---

## What does `ss` do?

Displays socket information.

---

## What does `ss -lntu` show?

Listening TCP and UDP sockets numerically.

---

## What does `systemctl --failed` show?

systemd units currently in failed state.

---

## Why doesn't one failed systemd unit automatically mean the whole server is critical?

The CLI lacks business context.

A failed optional test service and a failed critical database service do not have equal operational impact.

---

## What does `journalctl -b` mean?

Restrict output to the current boot.

---

## What does `journalctl -p err` mean?

Filter by error priority and more severe priorities.

---

## Why shouldn't the CLI automatically run with sudo?

Most information does not require root. Automatic privilege escalation violates least-privilege principles and can increase risk.

---

## Why validate arguments?

To prevent incorrect behavior, confusing errors, and unsafe interpretation of user input.

---

## Why validate configuration?

Configuration is also input.

Invalid configuration can cause a correctly written program to behave incorrectly.

---

## What is semantic validation?

Checking whether a value makes logical sense, not merely whether it has the correct data type.

Example:

```text
warning=95
critical=80
```

contains valid integers but invalid threshold ordering.

---

## Why avoid `eval`?

`eval` executes constructed shell code and can introduce severe command-injection risks when combined with untrusted input.

---

## Why is `source` potentially dangerous?

Sourcing a file executes its shell contents.

Only trusted configuration should be sourced.

---

## What is the difference between collection and interpretation?

Collection obtains facts:

```text
Filesystem / is 82% used
```

Interpretation applies meaning:

```text
82% exceeds warning threshold
```

Keeping these concepts separate improves tool design.

---

## Why have an `all` command and a `diagnose` command?

`all` provides broad system information.

`diagnose` emphasizes information commonly useful during troubleshooting.

---

```text
ShellCheck
CI

```
